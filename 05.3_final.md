import pandas as pd
import numpy as np

def impute_missing_values(df: pd.DataFrame, strategy: str = "mean") -> pd.DataFrame:
    """
    Imputes missing values in a Pandas DataFrame using the specified strategy.

    Parameters:
    df (pd.DataFrame): DataFrame to impute missing values in.
    strategy (str, optional): Imputation strategy. Defaults to "mean".
        - "mean": Impute with the mean of each column.
        - "median": Impute with the median of each column.
        - "most_frequent": Impute with the most frequent value in each column.

    Returns:
    pd.DataFrame: DataFrame with imputed missing values.

    Assumptions:
    - The columns contain numeric data for "mean" and "median" strategies.
    - The columns contain categorical data for "most_frequent" strategy.
    """
    if not isinstance(df, pd.DataFrame):
        raise ValueError("Input must be a Pandas DataFrame")
    if df.empty:
        raise ValueError("Input DataFrame is empty")

    imputed_df = df.copy()

    for col in imputed_df.columns:
        if strategy == "mean":
            if pd.api.types.is_numeric_dtype(imputed_df[col]):
                mean_val = imputed_df[col].mean(skipna=True)
                imputed_df[col] = imputed_df[col].fillna(mean_val)
        elif strategy == "median":
            if pd.api.types.is_numeric_dtype(imputed_df[col]):
                median_val = imputed_df[col].median(skipna=True)
                imputed_df[col] = imputed_df[col].fillna(median_val)
        elif strategy == "most_frequent":
            most_frequent_val = imputed_df[col].mode().iloc[0]
            imputed_df[col] = imputed_df[col].fillna(most_frequent_val)
        else:
            raise ValueError("Invalid imputation strategy")

    return imputed_df
